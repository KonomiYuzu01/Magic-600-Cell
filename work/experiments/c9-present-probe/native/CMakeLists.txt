cmake_minimum_required(VERSION 3.20)
project(c9_present_probe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The C9 present probe requires MSVC and the Windows SDK")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
# No shader compiler: the core layer draws only ClearRenderTargetView rectangles.
add_library(c9pp SHARED src/c9pp.cpp src/code.cpp)
target_compile_definitions(c9pp PRIVATE C9PP_EXPORTS)
target_link_libraries(c9pp PRIVATE d3d12 dxgi dcomp user32 gdi32)
add_executable(c9pp_control src/control.cpp)
target_link_libraries(c9pp_control PRIVATE c9pp d3d12 dxgi dcomp user32 gdi32)
foreach(target c9pp c9pp_control)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
        _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
endforeach()

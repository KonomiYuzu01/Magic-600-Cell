cmake_minimum_required(VERSION 3.20)
project(sb_handoff LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "sb_handoff requires the Windows SDK")
endif()
add_executable(sb_handoff src/main.cpp src/protocol.cpp src/handoff.cpp)
target_compile_features(sb_handoff PRIVATE cxx_std_20)
target_compile_definitions(sb_handoff PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sb_handoff PRIVATE /W4 /EHsc /permissive- /utf-8)
target_link_libraries(sb_handoff PRIVATE d3d12 dxgi d3d11 dxguid)

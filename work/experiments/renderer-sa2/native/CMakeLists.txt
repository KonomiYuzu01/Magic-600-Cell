cmake_minimum_required(VERSION 3.20)
project(sa2_native LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "SA2 native requires MSVC and the Windows SDK")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
add_library(sa2_pure STATIC src/pure.cpp)
target_include_directories(sa2_pure PUBLIC include src)
add_library(sa2_interop SHARED src/interop.cpp)
target_compile_definitions(sa2_interop PRIVATE SA2_INTEROP_EXPORTS)
target_link_libraries(sa2_interop PRIVATE sa2_pure d3d12 dxgi dxguid)
add_executable(sa2_selftest src/selftest.cpp src/gpu_test.cpp)
target_link_libraries(sa2_selftest PRIVATE sa2_pure sa2_interop d3d12 dxgi dxguid)
foreach(target sa2_pure sa2_interop sa2_selftest)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
endforeach()

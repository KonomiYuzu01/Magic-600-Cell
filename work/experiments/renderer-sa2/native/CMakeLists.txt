cmake_minimum_required(VERSION 3.20)
project(sa2_native LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "SA2 native requires MSVC and the Windows SDK")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(SB_PROBE "${REPO_ROOT}/work/experiments/renderer-sb/probe")
find_program(DXC_EXECUTABLE NAMES dxc REQUIRED)
# Compile the readers/checks byte-identical. Rename only their unused standalone
# entry point so the self-test can link the same library without a second main.
add_library(sa2_sb STATIC "${SB_PROBE}/src/probe.cpp" "${SB_PROBE}/src/edges.cpp" src/sb_link.cpp)
target_include_directories(sa2_sb PUBLIC "${SB_PROBE}/src")
target_compile_definitions(sa2_sb PRIVATE M600_REPO_ROOT="${REPO_ROOT}" main=sa2_unused_sb_main)
target_link_libraries(sa2_sb PUBLIC bcrypt)
add_library(sa2_pure STATIC src/pure.cpp)
target_include_directories(sa2_pure PUBLIC include src)
add_library(sa2_scene_record STATIC src/scene_record.cpp)
target_include_directories(sa2_scene_record PUBLIC include src)
target_link_libraries(sa2_scene_record PUBLIC sa2_sb)
add_library(sa2_interop SHARED src/interop.cpp src/scene.cpp)
target_compile_definitions(sa2_interop PRIVATE SA2_INTEROP_EXPORTS)
target_link_libraries(sa2_interop PRIVATE sa2_pure sa2_scene_record d3d12 dxgi dxguid)
add_executable(sa2_selftest src/selftest.cpp src/gpu_test.cpp)
target_link_libraries(sa2_selftest PRIVATE sa2_pure sa2_scene_record sa2_interop d3d12 dxgi dxguid)
foreach(target sa2_sb sa2_pure sa2_scene_record sa2_interop sa2_selftest)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
endforeach()
function(compile_shader output source entry profile)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${output}"
        COMMAND "${DXC_EXECUTABLE}" -nologo -O3 -Ges -T "${profile}" -E "${entry}"
            ${ARGN} -I "${SB_PROBE}/shaders" -Fo "${CMAKE_CURRENT_BINARY_DIR}/${output}"
            "${SB_PROBE}/shaders/${source}"
        DEPENDS "${SB_PROBE}/shaders/${source}" "${SB_PROBE}/shaders/geometry.hlsl" VERBATIM)
endfunction()
compile_shader(draw_vs.dxil draw.hlsl vsMain vs_6_0)
compile_shader(draw_ps.dxil draw.hlsl psMain ps_6_0)
compile_shader(count_vs.dxil draw.hlsl vsMain vs_6_0 -DCOUNT_INVOCATIONS=1)
compile_shader(geometry_cs.dxil check.hlsl csMain cs_6_0)
add_custom_target(sa2_shaders ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/draw_vs.dxil"
    "${CMAKE_CURRENT_BINARY_DIR}/draw_ps.dxil" "${CMAKE_CURRENT_BINARY_DIR}/count_vs.dxil"
    "${CMAKE_CURRENT_BINARY_DIR}/geometry_cs.dxil")
add_dependencies(sa2_interop sa2_shaders)
